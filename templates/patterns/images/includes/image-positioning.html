Using columns for image placement

Images will never exceed 100% of the width of their parent element, but will not scale up if the parent is wider than the image.

{% for class in ["third", "half", "twothird"] %} {% if class == "third" %} {% set subclass = "twothird" %} {% elseif class == "twothird" %} {% set subclass = "third" %} {% else %} {% set subclass = "half" %} {% endif %}

.{{ class }}/.{{ subclass }}

.{{ subclass }}


<div class="{{ class }}">
    <figure>
        <img src="/assets/images/examples/example.jpg" alt="Example image.">
        <figcaption><samp>.{{ class }}</samp></figcaption>
    </figure>
</div><!-- /{{ class }} -->
<div class="{{ subclass }}">
    <p><samp>.{{ subclass }}</samp></p>
</div><!-- /{{ subclass }} -->
{% endfor %}

</div><!-- /span-{{ x }} -->
