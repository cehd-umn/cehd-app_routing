To add a background with the correct shade of blue and bigger margins to one of these, add the .emphasis and .subtle classes.


<div class="row emphasis margin-top-1">
    <div class="container">
        <div class="half feature">
            <p>A half-width column with yellow background</p>
        </div>
    </div>
</div>

Sass



// public/assets/src/sass/_colors.scss
$base-white: #fff;
$cehd-tertiary-gold: #ffec99;
$cehd-light-grey: rgb(239, 239, 238);
$umn-light-blue-background: #d0edf5;
$umn-light-brown-foreground: #91785b;

// public/assets/src/sass/_core.scss
.emphasis {
    background-color: $cehd-tertiary-gold;
    padding: 2em;

    &.subtle {
        background-color: $cehd-light-grey;

        .button {
            border-color: $base-white;
        }
    }

    &.subtle {
        background-color: $umn-light-blue-background;
    }

    &.earthy {
        background-color: $umn-light-brown-foreground;
    }
}

Rendered CSS


.emphasis {
    background-color: #ffec99;
    padding: 2em;
}

/* The subtle class overrides the default background color. */
.emphasis.subtle {
    background-color: rgb(239, 239, 238);
}
